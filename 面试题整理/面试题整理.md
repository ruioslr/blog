### 树的bfs
使用队列来遍历整棵树
```js
function bfs(root){
    const queue = [root];
    
    const result = [];
    
    while(queue.length > 0){
     const node = queue.shift();
     result.push(node);
     queue.push(...node.children);
    }
    return result;
}

```

### 树的dfs
```js
function dfs(root){
    const result = [];
    
    function traverse (node) {
        result.push(node);
        if(node.children.length){
            node.children.forEach(traverse)
        }
    }
    
    traverse(root);
    
    return result;
}
```

function debounce(fn, wait) {
    let timer = null;
    return function(...args) {
        const context = this;
        if(timer){
            clearTimeout(timer);
            timer = null;
        }
        timer = setTimeout(() => {
            fn.apply(context, args)
        },wait)
        
    }
}

function throttle (fun, wait){
  let timer = null;
  return function (...args){
      const ctx = this;
      if(!timer){
        timer =  setTimeout(() => {
            fun.apply(ctx, args)
           timer = null;
         }, wait)
      }
  }
}

### 实现一个lru
```js
class LRUCache {
    /**
     * @param {number} capacity 缓存的最大容量
     */
    constructor(capacity) {
        if (capacity <= 0) {
            throw new Error("缓存容量必须大于 0");
        }
        this.capacity = capacity;
        // 使用 Map 来存储缓存数据。Map 会记住键的插入顺序。
        // 当一个键被重新设置时，它会被移动到 Map 的末尾（最新使用）。
        this.cache = new Map();
    }

    /**
     * 获取指定 key 的值。
     * 如果 key 存在于缓存中，则返回其值，并将其标记为最近使用（移动到 Map 末尾）；
     * 否则返回 -1。
     * @param {number} key
     * @return {number}
     */
    get(key) {
        if (this.cache.has(key)) {
            const value = this.cache.get(key);
            // 将当前访问的 key 重新设置一次，使其移动到 Map 的末尾，表示最近使用
            this.cache.delete(key);
            this.cache.set(key, value);
            return value;
        }
        return -1;
    }

    /**
     * 将键值对放入缓存。
     * 如果 key 已存在，则更新其值并将其标记为最近使用；
     * 如果 key 不存在：
     * 如果缓存已满，则移除 Map 的第一个元素（最久未使用），然后添加新元素。
     * 如果缓存未满，则直接添加新元素。
     * @param {number} key
     * @param {number} value
     * @return {void}
     */
    put(key, value) {
        if (this.cache.has(key)) {
            // 如果 key 已经存在，先删除旧的，再重新设置，使其成为最新使用
            this.cache.delete(key);
        } else if (this.cache.size >= this.capacity) {
            // 如果缓存已满，删除 Map 中的第一个元素 (最久未使用的)
            // Map.keys().next().value 获取 Map 的第一个键
            const oldestKey = this.cache.keys().next().value;
            this.cache.delete(oldestKey);
        }
        // 添加或更新键值对，新添加/更新的键总是在 Map 的末尾
        this.cache.set(key, value);
    }
}
```

### 深拷贝
```js
// 深拷贝的实现
function deepCopy(object) {
  if (!object || typeof object !== "object") return;

  let newObject = Array.isArray(object) ? [] : {};

  for (let key in object) {
    if (object.hasOwnProperty(key)) {
      newObject[key] =
        typeof object[key] === "object" ? deepCopy(object[key]) : object[key];
    }
  }

  return newObject;
}
```

function compose(funcs){
    if(funcs.lenght === 0){
        return (args) => args; 
    }
    
    if(funcs.lenght === 1){
